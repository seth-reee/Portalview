cmake_minimum_required(VERSION 3.24)
project(Portalview VERSION 0.1.2 LANGUAGES CXX)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2 Widgets DBus)
qt_policy(SET QTP0001 NEW)
qt_policy(SET QTP0004 NEW)
qt_add_executable(portalview src/main.cpp src/manager.h src/manager.cpp src/theme.h src/tray.h src/tray.cpp src/singleinstance.h)
set_source_files_properties(Portalview.png PROPERTIES QT_RESOURCE_ALIAS portalview.png)
qt_add_resources(portalview "icons" PREFIX "/icons" FILES Portalview.png)
qt_add_resources(portalview "license" PREFIX "/" FILES LICENSE)
target_compile_definitions(portalview PRIVATE PORTALVIEW_VERSION="${PROJECT_VERSION}")
qt_add_qml_module(portalview URI Portalview VERSION 1.0 QML_FILES qml/Main.qml)
target_link_libraries(portalview PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::DBus)
install(TARGETS portalview RUNTIME DESTINATION bin)
configure_file(packaging/portalview.desktop ${CMAKE_CURRENT_BINARY_DIR}/portalview.desktop @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/portalview.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES Portalview.png DESTINATION share/pixmaps RENAME portalview.png)
install(FILES LICENSE DESTINATION share/licenses/portalview)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
    qt_add_executable(portalview_instance_tests tests/instance_test.cpp src/singleinstance.h)
    target_include_directories(portalview_instance_tests PRIVATE src)
    target_link_libraries(portalview_instance_tests PRIVATE Qt6::Test Qt6::DBus)
    add_test(NAME single_instance COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:portalview_instance_tests>)
    qt_add_executable(portalview_tests tests/manager_test.cpp src/manager.cpp src/manager.h)
    target_include_directories(portalview_tests PRIVATE src)
    target_link_libraries(portalview_tests PRIVATE Qt6::Test)
    add_test(NAME connections COMMAND portalview_tests)
    set_tests_properties(connections PROPERTIES ENVIRONMENT "XDG_DATA_HOME=${CMAKE_BINARY_DIR}/test-data")
    qt_add_executable(portalview_ui_tests tests/dialog_test.cpp src/manager.cpp src/manager.h src/theme.h src/tray.cpp src/tray.h)
    qt_add_resources(portalview_ui_tests "icons" PREFIX "/icons" FILES Portalview.png)
    qt_add_resources(portalview_ui_tests "license" PREFIX "/" FILES LICENSE)
    target_include_directories(portalview_ui_tests PRIVATE src)
    target_link_libraries(portalview_ui_tests PRIVATE Qt6::Test Qt6::Quick Qt6::QuickControls2 Qt6::Widgets)
    target_compile_definitions(portalview_ui_tests PRIVATE PORTALVIEW_QML_PATH="${CMAKE_CURRENT_SOURCE_DIR}/qml/Main.qml")
    add_test(NAME dialogs COMMAND portalview_ui_tests)
    set_tests_properties(dialogs PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;XDG_DATA_HOME=${CMAKE_BINARY_DIR}/test-data;XDG_CONFIG_HOME=${CMAKE_BINARY_DIR}/test-data/config")
endif()
